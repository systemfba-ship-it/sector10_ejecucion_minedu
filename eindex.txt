function fmtMoney(n){
  return "S/ " + (n||0).toLocaleString('es-PE', {maximumFractionDigits:0});
}
function fmtCompact(n){
  n = n || 0;
  if (n >= 1e9) return "S/ " + (n/1e9).toFixed(1) + " mil M";
  if (n >= 1e6) return "S/ " + (n/1e6).toFixed(1) + " M";
  if (n >= 1e3) return "S/ " + (n/1e3).toFixed(1) + " mil";
  return "S/ " + n.toFixed(0);
}
function pctColor(p){
  if (p >= 60) return "var(--good)";
  if (p >= 35) return "var(--mid)";
  return "var(--low)";
}
function barCell(pct){
  const p = Math.max(0, Math.min(100, pct||0));
  return `<div class="bar-wrap">
      <div class="bar-track"><div class="bar-fill" style="width:${p}%; background:${pctColor(p)}"></div></div>
      <span class="pct-label" style="color:${pctColor(p)}">${p.toFixed(1)}%</span>
    </div>`;
}

const map = L.map('map', {scrollWheelZoom:false, zoomControl:true}).setView([-9.2, -75.5], 5);
L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', {
  attribution: '',
  maxZoom: 8
}).addTo(map);

let geoLayer = null;

function colorForPct(devengado, maxDev){
  const t = maxDev > 0 ? Math.min(1, devengado / maxDev) : 0;
  const stops = [
    {t:0,   c:[27,58,46]},
    {t:0.5, c:[47,120,90]},
    {t:1,   c:[46,204,113]}
  ];
  let a = stops[0], b = stops[1];
  if (t > 0.5){ a = stops[1]; b = stops[2]; }
  const localT = a.t === b.t ? 0 : (t - a.t) / (b.t - a.t);
  const rgb = a.c.map((v,i) => Math.round(v + (b.c[i]-v)*localT));
  return `rgb(${rgb[0]},${rgb[1]},${rgb[2]})`;
}

function renderMap(){
  const rows = DATA_REGIONES.filter(r => r.ANO_EJE === currentYear);
  const byCode = {};
  rows.forEach(r => byCode[r.DEPARTAMENTO_META] = r);
  const maxDev = Math.max(0, ...rows.map(r => r.devengado || 0));

  if (geoLayer) map.removeLayer(geoLayer);

  geoLayer = L.geoJSON(PERU_GEOJSON, {
    style: feat => {
      const code = feat.properties.FIRST_IDDP;
      const r = byCode[code];
      return {
        fillColor: colorForPct(r ? r.devengado : 0, maxDev),
        fillOpacity: 0.85,
        color: '#0b0c10',
        weight: 1
      };
    },
    onEachFeature: (feat, layer) => {
      const code = feat.properties.FIRST_IDDP;
      const r = byCode[code];
      const nombre = feat.properties.NOMBDEP;
      const html = r
        ? `<strong>${nombre}</strong><br>PIM: ${fmtMoney(r.pim)}<br>Devengado: ${fmtMoney(r.devengado)}<br>Avance: ${r.avance_pct.toFixed(1)}%`
        : `<strong>${nombre}</strong><br>Sin datos`;
      layer.bindTooltip(html, {className:'region-tooltip', sticky:true});
      layer.on('mouseover', () => layer.setStyle({weight:2, color:'#4ea8de'}));
      layer.on('mouseout', () => layer.setStyle({weight:1, color:'#0b0c10'}));
    }
  }).addTo(map);
}

const tbodyP = document.querySelector('#tbl-pliegos tbody');
const searchP = document.getElementById('search-pliego');
const sortP = document.getElementById('sort-pliego');
const countP = document.getElementById('pliegos-count');

function renderPliegos(){
  let rows = DATA_PLIEGOS.filter(r => r.ANO_EJE === currentYear);
  const total = rows.length;
  const q = searchP.value.trim().toLowerCase();
  if (q) rows = rows.filter(r => r.PLIEGO_NOMBRE.toLowerCase().includes(q) || r.PLIEGO.includes(q));

  const sort = sortP.value;
  rows.sort((a,b) => {
    if (sort === 'pim-desc') return b.pim - a.pim;
    if (sort === 'pim-asc') return a.pim - b.pim;
    if (sort === 'avance-desc') return b.avance_pct - a.avance_pct;
    if (sort === 'avance-asc') return a.avance_pct - b.avance_pct;
    if (sort === 'nombre-asc') return a.PLIEGO_NOMBRE.localeCompare(b.PLIEGO_NOMBRE);
    return 0;
  });

  countP.textContent = `${rows.length} de ${total} pliegos`;

  if (rows.length === 0){
    tbodyP.innerHTML = `<tr><td colspan="5" class="empty">Sin resultados</td></tr>`;
    return;
  }

  tbodyP.innerHTML = rows.map(r => `
    <tr data-pliego="${r.PLIEGO}" class="${activePliego === r.PLIEGO ? 'active' : ''}">
      <td>${r.PLIEGO_NOMBRE}<div style="font-size:11px;color:var(--muted)">Pliego ${r.PLIEGO}</div></td>
      <td class="num">${fmtMoney(r.pia)}</td>
      <td class="num">${fmtMoney(r.pim)}</td>
      <td class="num">${fmtMoney(r.devengado)}</td>
      <td class="num">${barCell(r.avance_pct)}</td>
    </tr>
  `).join('');

  tbodyP.querySelectorAll('tr[data-pliego]').forEach(tr => {
    tr.addEventListener('click', () => {
      const code = tr.getAttribute('data-pliego');
      activePliego = (activePliego === code) ? null : code;
      renderPliegos();
      renderEjecutoras();
    });
  });
}

searchP.addEventListener('input', renderPliegos);
sortP.addEventListener('change', renderPliegos);

function renderEjecutoras(){

  const total = rows.length;

  if (activePliego){
    rows = rows.filter(r => r.PLIEGO === activePliego);

    tagEl.innerHTML = `<span class="pliego-tag">${pliegoNombre}</span> <button class="clear-filter" id="clear-filter">quitar filtro</button>`;
    document.getElementById('clear-filter').addEventListener('click', () => {
      activePliego = null;
      renderPliegos();
      renderEjecutoras();
    });
  } else {
    tagEl.innerHTML = '';
  }

  const q = searchE.value.trim().toLowerCase();
  if (q) rows = rows.filter(r => r.EJECUTORA_NOMBRE.toLowerCase().includes(q));

  const sort = sortE.value;
  rows.sort((a,b) => {
    if (sort === 'pim-desc') return b.pim - a.pim;
    if (sort === 'pim-asc') return a.pim - b.pim;
    if (sort === 'avance-desc') return b.avance_pct - a.avance_pct;
    if (sort === 'avance-asc') return a.avance_pct - b.avance_pct;
    if (sort === 'nombre-asc') return a.EJECUTORA_NOMBRE.localeCompare(b.EJECUTORA_NOMBRE);
    return 0;
  });

  countE.textContent = `${rows.length} de ${total} ejecutoras`;

  if (rows.length === 0){
    tbodyE.innerHTML = `<tr><td colspan="6" class="empty">Sin resultados</td></tr>`;
    return;
  }

  tbodyE.innerHTML = rows.map(r => {
    const pliegoNombre = (DATA_PLIEGOS.find(p => p.PLIEGO === r.PLIEGO && p.ANO_EJE === currentYear) || {}).PLIEGO_NOMBRE || r.PLIEGO;
    return `
    <tr>
      <td>${r.EJECUTORA_NOMBRE}<div style="font-size:11px;color:var(--muted)">Sec. ejec. ${r.SEC_EJEC}</div></td>
      <td>${pliegoNombre}</td>
      <td class="num">${fmtMoney(r.pia)}</td>
      <td class="num">${fmtMoney(r.pim)}</td>
      <td class="num">${fmtMoney(r.devengado)}</td>
      <td class="num">${barCell(r.avance_pct)}</td>
    </tr>`;
  }).join('');
}

searchE.addEventListener('input', renderEjecutoras);
